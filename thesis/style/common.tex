% ---- Look ------------------------------------------------------------
% One source, two looks. The content and the page order are the same.
%   \latexlooktrue   (default) the typography of a LaTeX research text:
%                    Computer Modern with Greek (CMU fonts), Latin Modern
%                    Math, single spacing, LaTeX's own headings, 3 cm
%                    margins, mixed-case headings, empty blank leaves.
%   \latexlookfalse  the Guidelines' Word look (A.2): Times New Roman
%                    11 pt, line spacing 1.5, headings 14/12 pt bold,
%                    2.54 cm margins, headings in capitals.
% The .docx always has the Word look (docx/make-reference.py).
\newif\iflatexlook
\latexlooktrue

% ---- Fonts ----------------------------------------------------------
\usepackage{fontspec}
\usepackage{polyglossia}

% LaTeX look: CMU (Computer Modern Unicode) is Computer Modern with the
% Greek alphabet, which Latin Modern does not have. TeX Live ships it as
% cm-unicode; on Debian/Ubuntu it is the fonts-cmu package. Without it
% the build falls back to the Word look and says so in the log.
\iflatexlook
  \IfFontExistsTF{CMU Serif}{
    \newcommand{\thesisserif}{CMU Serif}
    \newcommand{\thesissans}{CMU Sans Serif}
    \newcommand{\thesismono}{CMU Typewriter Text}
    \newcommand{\thesismonoscale}{1.0}
  }{
    \PackageWarningNoLine{thesis}{CMU fonts not found, so the Word look
      is used. Install fonts-cmu (Debian/Ubuntu) or cm-unicode (TeX Live)}
    \latexlookfalse
  }
\fi

% Word look: the department requires Times New Roman at 11 pt
% (Guidelines A.2). Times New Roman itself is not redistributable, so the
% build asks for it first and falls back to Liberation Serif, which is
% metric-compatible: same widths, same line breaks, same page count.
\iflatexlook\else
  \IfFontExistsTF{Times New Roman}{
    \newcommand{\thesisserif}{Times New Roman}
    \newcommand{\thesissans}{Arial}
    \newcommand{\thesismono}{Courier New}
  }{
    \newcommand{\thesisserif}{Liberation Serif}
    \newcommand{\thesissans}{Liberation Sans}
    \newcommand{\thesismono}{Liberation Mono}
  }
  \newcommand{\thesismonoscale}{0.90}
\fi
\setmainfont[Ligatures=TeX]{\thesisserif}
\setsansfont[Ligatures=TeX]{\thesissans}
\setmonofont[Scale=\thesismonoscale,Ligatures=TeX]{\thesismono}

% ---- Maths -----------------------------------------------------------
% amssymb is NOT loaded: unicode-math (below) provides every
% symbol it defines, and loading both makes them fight over
% \eth, \digamma, \smallsetminus and \backepsilon.
\usepackage{amsmath}

% The maths FONT.
% Without this the text is Times (or its metric twin) and the
% mathematics is Computer Modern, because that is what XeLaTeX uses when
% nothing says otherwise. The two do not match, and the mismatch shows
% wherever a symbol sits inside a sentence: "FaB needs $5f+1$ nodes"
% printed the words in one face and the 5f+1 in another.
%
% TeX Gyre Termes Math is the companion of Times: same proportions, same
% weight, designed for exactly this pairing. If it is missing the
% document still builds, in Computer Modern maths, and says so in the
% log rather than failing.
%
% NOT the text font: TeX Gyre Termes has no accented Greek — no ά, έ, ή,
% ί, ό, ύ, ώ — so the Greek edition would be full of missing glyphs.
% Liberation Serif has the full range, which is why it is the fallback.
\usepackage{unicode-math}
\iflatexlook
  % Latin Modern Math is the maths of Computer Modern: the pairing that
  % LaTeX papers use. TeX convention: Latin and lowercase Greek italic,
  % Greek capitals upright (unicode-math's default math-style).
  \setmathfont{latinmodern-math.otf}
\else
\IfFontExistsTF{texgyretermes-math.otf}{
  \setmathfont{texgyretermes-math.otf}
  % unicode-math sets upright Greek capitals by ISO style; LaTeX's own
  % convention, and every source this thesis quotes, prints them upright.
  \setmathfont[range=\mathup/{greek,Greek}]{texgyretermes-math.otf}
}{
  \PackageWarningNoLine{thesis}{TeX Gyre Termes Math not found —
    mathematics will be Computer Modern and will not match the text.
    Install the tex-gyre-math package}
}
\fi

% ---- Source-quality markers ------------------------------------------
% Project rule R9: a source that has not been through peer review is
% marked WHERE IT IS USED, not only in the bibliography. A reader
% checking a claim should not have to turn to the back to find out that
% the evidence is a preprint.
% Capitals, not small capitals: Times New Roman has no true small-caps
% face, and neither does the metric-compatible fallback, so \textsc{}
% only produced a font warning and a synthesised shape.
\newcommand{\grey}{\textsuperscript{G}}   % grey literature
\newcommand{\prep}{\textsuperscript{P}}   % preprint, not refereed
% U marks a DIFFERENT thing from G and P. Those two say what kind of
% publication a source is. U says that the thesis could not verify the
% source's own bibliographic data against the article or an index, and
% took it from another work's reference list. A reader who wants to
% follow the citation needs to know that before they try.
\newcommand{\unver}{\textsuperscript{U}}   % bibliographic data unverified

% ---- Headings --------------------------------------------------------
% Guidelines A.2: chapter headings 14 pt bold with 12 pt before and
% after; sub-headings 12 pt bold; every chapter on a new page. The
% template sets a chapter as two lines: "ΚΕΦΑΛΑΙΟ 1" flush left, the
% title centred under it. docx/make-reference.py gives the .docx the
% same (Chapter Label: left; Heading 1: centred, 14 pt).
% LaTeX look: the report class's own headings, restated for titlesec.
\usepackage{titlesec}
\iflatexlook
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries}{\chaptertitlename\ \thechapter}{20pt}{\Huge}
\titlespacing*{\chapter}{0pt}{50pt}{40pt}
\else
\titleformat{\chapter}[display]
  {\bfseries\fontsize{14}{17}\selectfont}
  {\filright\chaptertitlename\ \thechapter}{6pt}{\filcenter}
\titlespacing*{\chapter}{0pt}{12pt}{12pt}
\titleformat{\section}
  {\bfseries\fontsize{12}{15}\selectfont}{\thesection}{0.8em}{}
\titleformat{\subsection}
  {\bfseries\fontsize{12}{15}\selectfont}{\thesubsection}{0.8em}{}
\fi

% ---- Front-matter heading -------------------------------------------
% An unnumbered \chapter* (set by titlesec above), so these headings sit
% at the same height and size as the contents, the lists and the
% references. The Word look prints #1, in capitals as the template has
% it; the LaTeX look prints #2, the mixed-case form.
%   #1 = as the template prints it   #2 = as in the table of contents
\iflatexlook
  \newcommand{\frontheadingtext}[2]{#2}
\else
  \newcommand{\frontheadingtext}[2]{#1}
\fi
\newcommand{\frontheading}[2]{%
  \chapter*{\frontheadingtext{#1}{#2}}%
  \phantomsection
  \addcontentsline{toc}{chapter}{#2}%
  \markboth{#2}{#2}}

% Blank leaf, as the template has between front-matter sections. It is
% counted, so both editions and the .docx agree page for page. The Word
% look numbers it and says it is blank; the LaTeX look leaves it empty.
\newcommand{\blankleaf}{%
  \clearpage
  \iflatexlook
    \thispagestyle{empty}\null
  \else
    \thispagestyle{plain}%
    \null\vfill\begin{center}\small\dgblankleaftext\end{center}\vfill\null
  \fi
  \clearpage}
